    min_len = criteria["min_length"]
    
    if password in forbidden or len(password) < min_len:
        return "Заборонений"
    
    has_digit = any(c.isdigit() for c in password)
    has_upper = any(c.isupper() for c in password)
    has_lower = any(c.islower() for c in password)
    has_special = any(c in SPECIAL_CHARS for c in password)

    meets_any = has_digit or has_upper or has_lower or has_special

    if meets_all and len(password) >= (min_len + 4) and is_unique:
        return "Дуже сильний"
    
    if meets_all and len(password) < (min_len + 4):
        return "Сильний"
    
    if meets_any:
        return "Середній"
    
    return "Слабкий"

ROLE_PERMISSIONS = {
    "admin": {"read", "write", "delete", "execute", "export"},
    "analyst": {"read", "write", "export"},
    "auditor": {"read", "export"},
    "guest": {"read"}
}

USERS = {
    "nazar_admin": {
        "role": "admin",
        "department": "Cybersecurity",
        "is_active": True,
        "clearance_level": 3
    },
    "ihor_analyst": {
        "role": "analyst",
        "department": "SOC",
        "is_active": True,
        "clearance_level": 2
    },
    "olena_auditor": {
        "role": "auditor",
        "department": "Audit",
        "is_active": False,
        "clearance_level": 2
    },
    "guest_user": {
        "role": "guest",
        "department": "Guest",
        "is_active": True,
        "clearance_level": 1
    }
}

RESOURCES = {
    "confidential_report.pdf": {"required_level": 2, "action": "read"},
    "security_logs.db": {"required_level": 3, "action": "delete"},
    "system_settings.conf": {"required_level": 3, "action": "write"},
    "public_info.txt": {"required_level": 1, "action": "read"}
}


def check_access(username: str, resource_name: str, requested_action: str) -> tuple[bool, str]:
    user = USERS.get(username)
    if not user:
        return False, f"Користувача '{username}' не знайдено."

    if not user["is_active"]:
        return False, f"Акаунт користувача '{username}' деактивовано."

    user_role = user["role"]
    allowed_actions = ROLE_PERMISSIONS.get(user_role, set())

    if requested_action not in allowed_actions:
        return False, f"Роль '{user_role}' не має дозволу на дію '{requested_action}'."

    resource = RESOURCES.get(resource_name)
    if resource:
        if user["clearance_level"] < resource["required_level"]:
            return False, f"Недостатній рівень доступу (потрібно {resource['required_level']}, є {user['clearance_level']})."

    return True, "Доступ дозволено."

LOG_PATH = os.path.join(DATA_DIR, "log.json")


def hash_password(password: str) -> str:
    """Хешування пароля за допомогою SHA-256 з «солонкою»."""
    salt = "lab01_secure_salt"
    return hashlib.sha256((password + salt).encode('utf-8')).hexdigest()


def init_database():
    """Ініціалізація SQLite бази даних та створення таблиці користувачів."""
    if os.path.exists(DB_PATH):
        os.remove(DB_PATH)
        
    conn = sqlite3.connect(DB_PATH)
    cursor = conn.cursor()
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS users (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            username TEXT UNIQUE NOT NULL,
            password_hash TEXT NOT NULL,
            role TEXT NOT NULL
        )
    """)
    
    # Додаємо тестових користувачів з хешованими паролями
    sample_users = [
        ("nazar_admin", hash_password("IoT@S3curity"), "admin"),
        ("ihor_analyst", hash_password("Blockchain@Pr0tect"), "analyst"),
        ("guest_user", hash_password("Edge@S3curity"), "guest")
    ]
    
    cursor.executemany("INSERT OR IGNORE INTO users (username, password_hash, role) VALUES (?, ?, ?)", sample_users)
    conn.commit()
    conn.close()


def log_event(username: str, action: str, status: str):
    """Логування подій у JSON файл."""
    log_entry = {
        "timestamp": datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
        "username": username,
        "action": action,
        "status": status
    }
    
    logs = []
    if os.path.exists(LOG_PATH):
        try:
            with open(LOG_PATH, "r", encoding="utf-8") as f:
                logs = json.load(f)
        except json.JSONDecodeError:
            logs = []
            
    logs.append(log_entry)
    with open(LOG_PATH, "w", encoding="utf-8") as f:
        json.dump(logs, f, indent=4, ensure_ascii=False)


def authenticate_user(username: str, password: str) -> bool:
    """Перевірка автентифікації користувача через БД та логування результату."""
    conn = sqlite3.connect(DB_PATH)
    cursor = conn.cursor()
    cursor.execute("SELECT password_hash FROM users WHERE username = ?", (username,))
    row = cursor.fetchone()
    conn.close()
    
    if not row:
        log_event(username, "login", "FAILED_USER_NOT_FOUND")
        return False
        
    stored_hash = row[0]
    input_hash = hash_password(password)
    
    if stored_hash == input_hash:
        log_event(username, "login", "SUCCESS")
        return True
    else:
        log_event(username, "login", "FAILED_INVALID_PASSWORD")
        return False

import sys

# Власний виняток для перевірки довжини пароля
class ValidationError(Exception):
    pass


def generate_hash(password: str, salt: str = "00015") -> str:
    """Генерація хешу (SHA-256 для варіанта 15) з персональною солонкою."""
    if password is None or salt is None or password == "" or salt == "":
        raise ValueError("Пароль або сіль не можуть бути порожніми.")
    
    # Мінімальна довжина пароля згідно з логікою Завдання 1 (мін. 8 символів)
    min_length = 8
    if len(password) < min_length:
        raise ValidationError(f"Пароль занадто короткий (мінімум {min_length} символів).")
        
    data = password + salt
    return hashlib.sha256(data.encode('utf-8')).hexdigest()


def log_event(func):
    """Декоратор для логування подій автентифікації у JSON-файл."""
    @wraps(func)
    def wrapper(*args, **kwargs):
        username = args[0] if args else kwargs.get("username", "unknown")
        result_status = "failure"
        try:
            res = func(*args, **kwargs)
            if res:
                result_status = "success"
            return res
        except (ValueError, ValidationError, FileNotFoundError, PermissionError, IOError) as e:
            result_status = f"failure ({type(e).__name__})"
            raise
        finally:
            log_entry = {
                "event": "login",
                "user": username,
                "result": result_status,
                "timestamp": datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
                "args": list(args[1:]),
                "kwargs": kwargs
            }
            
            logs = []
            if os.path.exists(LOG_PATH):
                try:
                    with open(LOG_PATH, "r", encoding="utf-8") as f:
                        logs = json.load(f)
                except (json.JSONDecodeError, FileNotFoundError):
                    logs = []
                    
            logs.append(log_entry)
            try:
                with open(LOG_PATH, "w", encoding="utf-8") as f:
                    json.dump(logs, f, indent=4, ensure_ascii=False)
            except IOError:
                pass
                
    return wrapper


def create_user(username: str, password: str) -> tuple[str, str]:
    """Створення пари (логін, хеш) з персональною солонкою."""
    hashed = generate_hash(password, salt=f"{VARIANT_NUMBER:05d}" if isinstance(VARIANT_NUMBER, int) else "00015")
    return username, hashed


def create_users(users_list: list[tuple[str, str]]):
    """Створення CSV-файлу бази даних користувачів з обробкою винятків."""
    try:
        os.makedirs(DATA_DIR, exist_ok=True)
        with open(CSV_PATH, "w", newline="", encoding="utf-8") as f:
            writer = csv.writer(f)
            for uname, pwd in users_list:
                username, hash_val = create_user(uname, pwd)
                writer.writerow([username, hash_val])
    except (PermissionError, IOError) as e:
        print(f"Помилка доступу або запису файлу: {e}")
        raise


@log_event
def login(username: str, password: str) -> bool:
    """Автентифікація користувача з перевірками та декоратором логування."""
    if not username or not password:
        raise ValueError("Логін або пароль не можуть бути порожніми.")

    users_db = []
    try:
        if not os.path.exists(CSV_PATH):
            raise FileNotFoundError("Файл бази даних користувачів не знайдено.")
            
        with open(CSV_PATH, "r", encoding="utf-8") as f:
            reader = csv.reader(f)
            for row in reader:
                if len(row) == 2:
                    users_db.append((row[0], row[1]))
    except (FileNotFoundError, PermissionError, IOError) as e:
        print(f"Помилка роботи з файлом бази даних: {e}")
        raise

    input_hash = generate_hash(password, salt=f"{VARIANT_NUMBER:05d}" if isinstance(VARIANT_NUMBER, int) else "00015")

    for db_user, db_hash in users_db:
        if db_user == username and db_hash == input_hash:
            return True

    return False
